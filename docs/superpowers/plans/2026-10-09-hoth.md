**Goal:** Every NPC on Hoth is a model, enforced by a test and a probe, then Hoth gets its own cast, the snowspeeder tow-cable fight and a living Echo Base.

**Architecture:** One resolver (`surface/cast.js`) decides how a kind is drawn; a site that says `cast: 'models'` never gets a built figure, only a retry, a stand-in model or nothing. A site's `uniforms` dress the galaxy war's troopers in the world's kit. Content stays data in `sites/ice.js`; new rules are pure files with tests.

**Tech Stack:** React, three.js, Vite, Vitest, Playwright (probe), Meshy API (Phase 2, through `scripts/meshy-galaxy.mjs`).

**Spec:** `docs/superpowers/specs/2026-10-09-hoth-design.md`

- [ ] **Step 3: Write `cast.js`**

```js
// How a kind of person, creature or machine on a world is drawn, in one
// place: a walker cut from its model, a crew model, a catalogue model, a
// figure built in code, or a prop that walks. A world whose site says
// `cast: 'models'` (Hoth) never gets a built one: its files are tried twice
// (placer.js's loadGlb drops a failed url from its cache, so the second is
// a fresh fetch), then a stand-in of the same silhouette, then nothing. The
// design: docs/superpowers/specs/2026-10-09-hoth-design.md, section 1.
//
// resolveFigure(kind, makers, { only, standIns, warn }) → fig | null
//   makers: { walker?, crew, model, built, prop }, each (kind) → fig | null
// STAND_INS[kind] → a kind with a model; markBuilt(fig) → fig, marked

// a kind's stand-in where its own file won't come: one of the same shape
export const STAND_INS = {
  rebel: 'hothtrooper',
  rebelpilot: 'hothtrooper',
  hothtrooper: 'rebel',
  snowtrooper: 'stormtrooper',
  rieekan: 'hothtrooper',
  torynfarr: 'rebel',
  veers: 'officer',
  lukehoth: 'luke',
  hanhoth: 'han',
  leiahoth: 'leia',
  droid: 'r2d2',
  twoonebee: 'r2d2',
  astromech2: 'r2d2',
  astromech3: 'r2d2',
  mercenary: 'rodian',
};

export const markBuilt = (fig) => {
  if (fig?.model) fig.model.userData.built = true;
  return fig ?? null;
};

const quiet = (fn, kind) => Promise.resolve(fn ? fn(kind) : null).catch(() => null);
const files = async (kind, makers) => (await quiet(makers.crew, kind)) ?? (await quiet(makers.model, kind));

export async function resolveFigure(kind, makers, { only = false, standIns = STAND_INS, warn = null } = {}) {
  const walker = await quiet(makers.walker, kind);
  if (walker) return walker;
  const own = await files(kind, makers);
  if (own) return own;
  if (!only) return (await quiet(makers.built, kind)) ?? (await quiet(makers.prop, kind));
  const again = await files(kind, makers);
  if (again) return again;
  const stand = standIns[kind];
  const sub = stand ? await files(stand, makers) : null;
  if (sub) return sub;
  warn?.(kind);
  return null;
}
```

export function garrisonLife(life, troops, uniforms = null) {
  if (!life || !troops) return life;
  let changed = false;
  const mapped = life.map((entry) => {
    // (a named person, a quest giver, anyone the site calls by id keeps their kind and words)
    if (entry.id || entry.named || entry.quest) return entry;
    const kind = troopKind(entry.kind, troops);
    if (kind === entry.kind) return entry;
    const dressed = dressOf(kind, uniforms);
    if (dressed === entry.kind) return entry;
    changed = true;
    const out = { ...entry, kind: dressed, name: TROOP_NAMES[dressed] ?? TROOP_NAMES[kind] };
    delete out.says;
    return out;
  });
  return changed ? mapped : life;
}

- [ ] **Step 2: Implement**

```js
// The snowspeeders' tow cable: harpooned to a walker's leg, the speeder
// flies round and round its legs till the cable wraps them and it trips.
// Pure, tested. The design: docs/superpowers/specs/2026-10-09-hoth-design.md,
// section 3.

const TAU = Math.PI * 2;
const wrap = (a) => ((a + Math.PI) % TAU + TAU) % TAU - Math.PI;

export function createCable({ legs, turns = 3, band = [4, 14], ceiling = 12, slack = 2, reach = 40 }) {
  let last = null;
  let swept = 0;
  let out = 0;
  let state = 'winding';
  return {
    step(dt, { x, z, y }) {
      if (state !== 'winding') return state;
      const dx = x - legs[0];
      const dz = z - legs[1];
      const d = Math.hypot(dx, dz);
      if (d > reach) return (state = 'cut');
      const a = Math.atan2(dz, dx);
      const inBand = d >= band[0] && d <= band[1] && y <= ceiling;
      if (inBand) {
        out = 0;
        if (last != null) swept += wrap(a - last);
      } else if ((out += dt) > slack) return (state = 'cut');
      last = a;
      if (Math.abs(swept) >= turns * TAU) state = 'tripped';
      return state;
    },
    swept: () => swept,
    turnsLeft: () => Math.max(0, turns - Math.abs(swept) / TAU),
  };
}
```
